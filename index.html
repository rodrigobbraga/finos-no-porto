function doGet() { const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); const sheet = spreadsheet.getSheetByName('main'); if (!sheet) { return createJsonResponse({ sucesso: false, erro: 'A folha "main" não existe.' }); } const values = sheet.getDataRange().getValues(); if (values.length < 2) { return createJsonResponse({ sucesso: true, total: 0, bares: [] }); } const headers = values[0]; const timezone = spreadsheet.getSpreadsheetTimeZone() || Session.getScriptTimeZone(); const bares = values .slice(1) .filter(row => row.some(cell => cell !== '')) .map(row => { const bar = {}; headers.forEach((header, index) => { bar[header] = row[index]; }); // Campos numéricos vazios permanecem como null, nunca como 0. bar.bar_id = toNumberOrNull(bar.bar_id); bar.latitude = toNumberOrNull(bar.latitude); bar.longitude = toNumberOrNull(bar.longitude); bar.preco_fino = toNumberOrNull(bar.preco_fino); bar.tamanho_cl = toNumberOrNull(bar.tamanho_cl); // Calcula o preço por litro apenas quando há preço e tamanho válidos. if ( bar.preco_fino !== null && bar.tamanho_cl !== null && bar.tamanho_cl > 0 ) { bar.preco_litro = bar.preco_fino / (bar.tamanho_cl / 100); } else { bar.preco_litro = null; } // Envia a data como ISO com o fuso horário da spreadsheet. // Evita que "yyyy-MM-dd" seja interpretado pelo browser como UTC. if (bar.atualizado_em instanceof Date && !Number.isNaN(bar.atualizado_em.getTime())) { bar.atualizado_em = Utilities.formatDate( bar.atualizado_em, timezone, "yyyy-MM-dd'T'HH:mm:ssXXX" ); } else if (bar.atualizado_em !== null && bar.atualizado_em !== undefined) { const textoData = String(bar.atualizado_em).trim(); if (/^\d{4}-\d{2}-\d{2}$/.test(textoData)) { const dataLocal = new Date(`${textoData}T00:00:00`); if (!Number.isNaN(dataLocal.getTime())) { bar.atualizado_em = Utilities.formatDate( dataLocal, timezone, "yyyy-MM-dd'T'HH:mm:ssXXX" ); } } } return bar; }); return createJsonResponse({ sucesso: true, total: bares.length, bares: bares }); } function doPost(e) { try { if (!e || !e.parameter) { return createJsonResponse({ sucesso: false, erro: 'Pedido inválido.' }); } const barId = toNumberOrNull(e.parameter.bar_id); const tipo = String(e.parameter.tipo || '').trim(); const tamanhoCl = toNumberOrNull(e.parameter.tamanho_cl); const precoReportadoTexto = String(e.parameter.preco_reportado || '').trim(); const precoReportado = precoReportadoTexto === '' ? null : toNumberOrNull(precoReportadoTexto); if (barId === null || !Number.isInteger(barId) || barId <= 0) { return createJsonResponse({ sucesso: false, erro: 'bar_id inválido.' }); } if (!['confirmacao', 'alteracao'].includes(tipo)) { return createJsonResponse({ sucesso: false, erro: 'Tipo de feedback inválido.' }); } if (![20, 25].includes(tamanhoCl)) { return createJsonResponse({ sucesso: false, erro: 'Tamanho inválido. Apenas 20 ou 25 cl são permitidos.' }); } if (tipo === 'alteracao') { if ( precoReportado === null || !Number.isFinite(precoReportado) || precoReportado <= 0 ) { return createJsonResponse({ sucesso: false, erro: 'Preço reportado inválido.' }); } } const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); const mainSheet = spreadsheet.getSheetByName('main'); const feedbackSheet = spreadsheet.getSheetByName('feedback'); if (!mainSheet) { return createJsonResponse({ sucesso: false, erro: 'A folha "main" não existe.' }); } if (!feedbackSheet) { return createJsonResponse({ sucesso: false, erro: 'A folha "feedback" não existe.' }); } const lock = LockService.getScriptLock(); lock.waitLock(5000); try { const feedbackValues = feedbackSheet.getDataRange().getValues(); const feedbackHeaders = feedbackValues[0] || []; const colFeedbackId = feedbackHeaders.indexOf('id_feedback'); const colBarId = feedbackHeaders.indexOf('bar_id'); const colTipo = feedbackHeaders.indexOf('tipo'); const colTamanho = feedbackHeaders.indexOf('tamanho_cl'); const colPreco = feedbackHeaders.indexOf('preco_reportado'); const colDataHora = feedbackHeaders.indexOf('data_hora'); const colEstado = feedbackHeaders.indexOf('estado'); const colComentario = feedbackHeaders.indexOf('comentario'); const colunasObrigatorias = [ ['id_feedback', colFeedbackId], ['bar_id', colBarId], ['tipo', colTipo], ['tamanho_cl', colTamanho], ['preco_reportado', colPreco], ['data_hora', colDataHora], ['estado', colEstado] ]; for (const [nome, indice] of colunasObrigatorias) { if (indice === -1) { return createJsonResponse({ sucesso: false, erro: `Coluna "${nome}" não encontrada na folha feedback.` }); } } let novoIdFeedback = 1; if (feedbackValues.length > 1) { const ids = feedbackValues .slice(1) .map(row => toNumberOrNull(row[colFeedbackId])) .filter(id => Number.isInteger(id) && id > 0); if (ids.length > 0) { novoIdFeedback = Math.max(...ids) + 1; } } const agora = new Date(); const estado = tipo === 'confirmacao' ? 'validado' : 'pendente'; const novaLinha = new Array(feedbackHeaders.length).fill(''); novaLinha[colFeedbackId] = novoIdFeedback; novaLinha[colBarId] = barId; novaLinha[colTipo] = tipo; novaLinha[colTamanho] = tamanhoCl; novaLinha[colPreco] = tipo === 'alteracao' ? precoReportado : ''; novaLinha[colDataHora] = agora; novaLinha[colEstado] = estado; if (colComentario !== -1) { novaLinha[colComentario] = ''; } feedbackSheet.appendRow(novaLinha); if (tipo === 'confirmacao') { atualizarDataVerificacao(mainSheet, barId, agora); } return createJsonResponse({ sucesso: true, mensagem: tipo === 'confirmacao' ? 'Preço confirmado com sucesso.' : 'Alteração enviada com sucesso.' }); } finally { lock.releaseLock(); } } catch (error) { return createJsonResponse({ sucesso: false, erro: error.message || 'Erro inesperado.' }); } } function atualizarDataVerificacao(sheet, barId, data) { const values = sheet.getDataRange().getValues(); if (values.length < 2) { throw new Error('Não existem bares na folha main.'); } const headers = values[0]; const colBarId = headers.indexOf('bar_id'); const colAtualizado = headers.indexOf('atualizado_em'); if (colBarId === -1) { throw new Error('Coluna "bar_id" não encontrada na main.'); } if (colAtualizado === -1) { throw new Error('Coluna "atualizado_em" não encontrada na main.'); } for (let i = 1; i < values.length; i++) { if (Number(values[i][colBarId]) === barId) { sheet.getRange(i + 1, colAtualizado + 1).setValue(data); return; } } throw new Error(`Bar com ID ${barId} não encontrado.`); } function toNumberOrNull(value) { if (value === null || value === undefined) { return null; } if (typeof value === 'string') { const texto = value.trim(); if (texto === '') { return null; } value = texto.replace(',', '.'); } const number = Number(value); return Number.isFinite(number) ? number : null; } function createJsonResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
